## 链表实现
> **时间复杂度：** **$O(1)$**
```cpp
//LinkedStackOfStrings.h
#include <string>
#include <utility>

struct Node
{
    std::string item;
    Node* next;
	
    Node() : item{}, next(nullptr) {}
    Node(std::string s) : item(std::move(s)), next(nullptr) {}
    //std::move() 在编译期将“左值”强制转换为“右值”，实现资源所有权的无拷贝转移
};

class LinkedStackOfStrings
{
public:
	LinkedStackOfStrings() : head(nullptr) {};
	~LinkedStackOfStrings();
	
	//禁止拷贝
	LinkedStackOfStrings(const LinkedStackOfStrings&) = delete;
	LinkedStackOfStrings& operator=(const LinkedStackOfStrings&) = delete;

    bool isEmpty();
    void push(std::string item);
    std::string pop();
private:
    Node* head = nullptr;
};
```

```cpp
//LinkedStackOfStrings.cpp
#include "LinkedStackOfStrings.h"
#include <stdexcept>

LinkedStackOfStrings::~LinkedStackOfStrings()
{
	while(head != nullptr)
	{
		Node* temp = head;
		head = head->next;
		delete temp;
	}
}

bool LinkedStackOfStrings::isEmpty()
{
    return head == nullptr;
}

void LinkedStackOfStrings::push(std::string item)
{
    Node* newNode = new Node(std::move(item)); //直接搬进去
    newNode->next = head;
    head = newNode;
}

std::string LinkedStackOfStrings::pop()
{
    if (head == nullptr)
    {
        throw std::runtime_error("LinkedStackOfStrings::pop(): Stack is empty");
    }
    Node* next = head->next;
    std::string str = std::move(head->item); //直接搬出来
    delete head;
    head = next;
    return str;
}
```
