
## 前言
在实际需求中，也有需要将数组打乱的需求，这个时候，就需要 **洗牌(shuffle)** 

## 代码
```cpp
#include <iostream>
#include <vector>
#include <random>

template <typename T>
void mySwap(std::vector<T>& arr, int a, int b)
{
    T temp = arr[a];
    arr[a] = arr[b];
    arr[b] = temp;
}

template <typename T>
void shuffle(std::vector<T>& arr, std::mt19937& gen)
{
    int n = arr.size();
    for(int i = 0; i < n; i++)
    {
        std::uniform_int_distribution<int> dis(0, i);
        int r = dis(gen);
        mySwap(arr, r, i);
    }

}

int main(void)
{
    std::random_device rd;
    std::mt19937 gen(rd());
    std::vector<int> arr = {1, 2, 3, 4, 5, 6};

    shuffle(arr, gen);

    for(int i = 0; i < arr.size(); i++)
    {
        std::cout << arr[i] << " ";
    }
    std::cout << std::endl;
    std::cin.get();
    return 0;
}
```
